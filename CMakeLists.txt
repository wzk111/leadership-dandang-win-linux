cmake_minimum_required(VERSION 3.16)
project(WorkSidekick VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Widgets Network Concurrent Test)
add_library(worksidekick_core src/core/Core.cpp)
target_include_directories(worksidekick_core PUBLIC src)
target_link_libraries(worksidekick_core PUBLIC Qt6::Core)
add_library(worksidekick_ai src/ai/OpenAIProvider.cpp src/ai/IAIProvider.h)
target_link_libraries(worksidekick_ai PUBLIC worksidekick_core Qt6::Network)
if(BUILD_TESTING)
  add_executable(test_core tests/TestCore.cpp)
  target_link_libraries(test_core PRIVATE worksidekick_core Qt6::Test)
  add_test(NAME core COMMAND test_core)
  add_executable(test_ai tests/TestAI.cpp)
  target_link_libraries(test_ai PRIVATE worksidekick_ai Qt6::Test)
  add_test(NAME ai COMMAND test_ai)
endif()
add_library(worksidekick_ui src/ui/ResultCard.cpp src/platform/ClipboardSelectionProvider.cpp)
target_link_libraries(worksidekick_ui PUBLIC worksidekick_core Qt6::Widgets)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SECRET REQUIRED IMPORTED_TARGET libsecret-1)
  add_library(worksidekick_platform src/platform/linux/LinuxSecretStore.cpp src/platform/ISecretStore.h)
  target_link_libraries(worksidekick_platform PUBLIC worksidekick_core Qt6::Concurrent PRIVATE PkgConfig::SECRET)
  if(BUILD_TESTING)
    add_executable(test_secret tests/TestSecret.cpp)
    target_link_libraries(test_secret PRIVATE worksidekick_platform Qt6::Test)
  endif()
endif()
if(BUILD_TESTING)
  add_executable(test_ui tests/TestUI.cpp)
  target_link_libraries(test_ui PRIVATE worksidekick_ui Qt6::Test)
  add_test(NAME ui COMMAND test_ui)
endif()
