cmake_minimum_required(VERSION 3.16)
project(WorkSidekick VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Widgets Network Concurrent)
if(BUILD_TESTING)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Test)
endif()
add_library(worksidekick_core src/core/Core.cpp src/core/SelectionResolver.cpp src/core/Profile.cpp src/core/FeaturePreferences.cpp)
target_include_directories(worksidekick_core PUBLIC src)
target_link_libraries(worksidekick_core PUBLIC Qt6::Core)
add_library(worksidekick_interfaces src/platform/ISecretStore.h src/platform/ISelectionMonitor.h src/platform/IGlobalShortcut.h)
set_target_properties(worksidekick_interfaces PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(worksidekick_interfaces PUBLIC worksidekick_core)
add_library(worksidekick_ai src/ai/OpenAIProvider.cpp src/ai/IAIProvider.h)
target_link_libraries(worksidekick_ai PUBLIC worksidekick_core Qt6::Network)
add_library(worksidekick_ui src/ui/ResultCard.cpp src/ui/WorkspaceWindow.cpp
  src/ui/SettingsWindow.cpp src/ui/SelectionDiagnostics.cpp src/ui/ActionBar.cpp src/ui/ActionBarPlacement.cpp src/ui/ManualActionPalette.cpp src/ui/ProfileWindow.cpp src/ui/ReplyComposer.cpp src/platform/ClipboardSelectionProvider.cpp)
target_link_libraries(worksidekick_ui PUBLIC worksidekick_interfaces Qt6::Widgets)
add_library(worksidekick_controller src/app/AppController.cpp)
target_link_libraries(worksidekick_controller PUBLIC worksidekick_ai worksidekick_interfaces)
add_library(worksidekick_application src/app/Application.cpp src/app/ApplicationOverlay.cpp src/app/ApplicationManual.cpp src/app/InstanceCoordinator.cpp)
target_link_libraries(worksidekick_application PUBLIC worksidekick_controller worksidekick_ui)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SECRET REQUIRED IMPORTED_TARGET libsecret-1)
  add_library(worksidekick_platform src/platform/linux/LinuxSecretStore.cpp src/platform/linux/PlatformFactory.cpp)
  target_link_libraries(worksidekick_platform PUBLIC worksidekick_interfaces Qt6::Concurrent PRIVATE PkgConfig::SECRET)
  add_executable(worksidekick src/main.cpp)
  target_link_libraries(worksidekick PRIVATE worksidekick_application worksidekick_platform)
  install(TARGETS worksidekick RUNTIME DESTINATION bin)
  install(FILES resources/worksidekick.desktop DESTINATION share/applications)
  if(BUILD_TESTING)
    add_executable(test_secret tests/TestSecret.cpp)
    target_link_libraries(test_secret PRIVATE worksidekick_platform Qt6::Test)
  endif()
elseif(WIN32)
  message(STATUS "M0: shared libraries/tests only on Windows; native application backend is deferred to M5.")
else()
  message(FATAL_ERROR "Supported M0 application target: Linux.")
endif()
if(BUILD_TESTING)
  foreach(name IN ITEMS Core AI UI Controller Flow ActionBar Overlay Manual ManualFlow Instance ProductCore Generation ProductWidgets ProductIntegration)
    string(TOLOWER "${name}" lower)
    add_executable(test_${lower} tests/Test${name}.cpp)
    target_link_libraries(test_${lower} PRIVATE worksidekick_application Qt6::Test)
    add_test(NAME ${lower} COMMAND test_${lower})
    set_tests_properties(${lower} PROPERTIES TIMEOUT 30)
  endforeach()
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  pkg_check_modules(ATSPI REQUIRED IMPORTED_TARGET atspi-2 gobject-2.0)
  add_library(worksidekick_selection src/platform/linux/AtSpiSelectionProvider.cpp
    src/platform/linux/AtSpiSelectionMonitor.cpp src/platform/linux/AtSpiNativeSession.cpp src/platform/linux/IAtSpiSession.h)
  target_link_libraries(worksidekick_selection PUBLIC worksidekick_interfaces PRIVATE PkgConfig::ATSPI)
  if(BUILD_TESTING)
    foreach(name IN ITEMS AtSpiSelection SelectionMonitor)
      add_executable(test_${name} tests/Test${name}.cpp)
      target_link_libraries(test_${name} PRIVATE worksidekick_selection Qt6::Test)
      add_test(NAME ${name} COMMAND test_${name})
      set_tests_properties(${name} PROPERTIES TIMEOUT 30)
    endforeach()
  endif()
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(worksidekick_platform PUBLIC worksidekick_selection)
  if(BUILD_TESTING)
    add_executable(accessible_fixture tests/AccessibleFixture.cpp)
    target_link_libraries(accessible_fixture PRIVATE Qt6::Widgets)
    add_executable(test_atspi_runtime tests/TestAtSpiRuntime.cpp)
    target_link_libraries(test_atspi_runtime PRIVATE worksidekick_selection Qt6::Test)
  endif()
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(worksidekick_window_policy src/platform/linux/LinuxWindowPolicy.cpp)
  target_link_libraries(worksidekick_window_policy PUBLIC worksidekick_ui)
  target_link_libraries(worksidekick_platform PUBLIC worksidekick_window_policy)
  if(BUILD_TESTING)
    find_package(X11 REQUIRED)
    add_executable(test_window_policy tests/TestWindowPolicy.cpp)
    target_link_libraries(test_window_policy PRIVATE worksidekick_window_policy Qt6::Test)
    add_test(NAME window_policy COMMAND test_window_policy)
    add_executable(test_overlay_runtime tests/TestOverlayRuntime.cpp)
    target_link_libraries(test_overlay_runtime PRIVATE worksidekick_application worksidekick_selection worksidekick_window_policy Qt6::Test X11::X11)
  endif()
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(X11 REQUIRED)
  find_package(Qt6 6.2 REQUIRED COMPONENTS DBus)
  add_library(worksidekick_shortcuts src/platform/linux/LinuxSelectionResolver.cpp
    src/platform/linux/X11GlobalShortcut.cpp src/platform/linux/PortalGlobalShortcut.cpp
    src/platform/linux/PortalTransport.h src/platform/linux/QtPortalTransport.cpp)
  target_link_libraries(worksidekick_shortcuts PUBLIC worksidekick_ui PRIVATE X11::X11 Qt6::DBus)
  target_link_libraries(worksidekick_platform PUBLIC worksidekick_shortcuts PRIVATE Qt6::DBus X11::X11)
  if(BUILD_TESTING)
    add_executable(test_portal tests/TestPortal.cpp)
    target_link_libraries(test_portal PRIVATE worksidekick_shortcuts Qt6::Test)
    add_test(NAME portal COMMAND test_portal)
    add_executable(test_x11_shortcut tests/TestX11Shortcut.cpp)
    target_link_libraries(test_x11_shortcut PRIVATE worksidekick_shortcuts Qt6::Test X11::X11 X11::Xtst)
    add_test(NAME x11_shortcut COMMAND test_x11_shortcut)
    set_tests_properties(portal x11_shortcut PROPERTIES TIMEOUT 30)
  endif()
endif()
if(BUILD_TESTING)
  add_executable(instance_peer tests/InstancePeer.cpp)
  target_link_libraries(instance_peer PRIVATE worksidekick_application)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND BUILD_TESTING)
  add_executable(test_portal_dbus tests/TestPortalDBus.cpp)
  target_link_libraries(test_portal_dbus PRIVATE worksidekick_shortcuts Qt6::DBus Qt6::Test)
  add_test(NAME portal_dbus COMMAND dbus-run-session -- $<TARGET_FILE:test_portal_dbus>)
  set_tests_properties(portal_dbus PROPERTIES TIMEOUT 30)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND BUILD_TESTING)
  add_executable(primary_owner tests/PrimaryOwner.cpp)
  target_link_libraries(primary_owner PRIVATE Qt6::Widgets)
endif()
