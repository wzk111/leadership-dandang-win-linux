cmake_minimum_required(VERSION 3.16)
project(WorkSidekick VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Widgets Network Concurrent Test)
add_library(worksidekick_core src/core/Core.cpp)
target_include_directories(worksidekick_core PUBLIC src)
target_link_libraries(worksidekick_core PUBLIC Qt6::Core)
if(BUILD_TESTING)
  add_executable(test_core tests/TestCore.cpp)
  target_link_libraries(test_core PRIVATE worksidekick_core Qt6::Test)
  add_test(NAME core COMMAND test_core)
endif()
